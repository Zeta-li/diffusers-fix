MODES = ["D", "V", "K", "B"]
ALIASES = {"D": "D", "V": "V", "K": "K", "B": "B", "Vonly": "V", "Konly": "K", "Both": "B"}
ERROR_FIELDS = ["mse", "rmse", "relative_rmse", "cosine", "squared_error_sum", "reference_squared_sum", "count"]


def mode_id(value):
    return ALIASES.get(str(value), str(value))


def read_jsonl(path):
    return [json.loads(line) for line in path.read_text().splitlines() if line.strip()] if path.exists() else []


def write_json(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False))


def save_csv(path, rows):
    if not rows:
        return
    fields = list(dict.fromkeys(key for row in rows for key in row))
    with path.open("w", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def finite(value):
    return value is not None and math.isfinite(float(value))


def display(value, digits=5):
    if isinstance(value, bool):
        return "是" if value else "否"
    if value is None:
        return "N/A"
    return f"{value:.{digits}g}" if isinstance(value, (int, float)) else str(value)


def metadata_value(metadata, keys):
    for key in keys:
        value = metadata.get(key, metadata.get("paths", {}).get(key))
        if value:
            return value
    return None


def resolve_path(value, base):
    path = Path(value).expanduser()
    return path.resolve() if path.is_absolute() else (base / path).resolve()


def normalize_steps(records):
    result = {}
    duplicates = []
    for row in records:
        if row.get("status") == "failed" or "active_error" not in row:
            continue
        key = (row["case_id"], mode_id(row["mode"]), int(row["step"]))
        if key in result:
            duplicates.append(list(key))
        result[key] = row
    return result, duplicates


def step_metrics(row):
    active = row["active_error"]
    inactive = row["inactive_error"]
    combined = row["combined_error"]
    a = float(active["squared_error_sum"])
    i = float(inactive["squared_error_sum"])
    total = float(combined["squared_error_sum"])
    delta = float(row["sigma_delta"])
    residual = total - a - i
    cross = row.get("orthogonality_cross_term")
    returned = row.get("returned_error", {})
    expected_returned_sse = {"V": i, "K": a, "B": total}.get(mode_id(row["mode"]))
    tolerance = 1e-5 * max(1.0, abs(total), a + i)
    full = bool(row["full_step"])
    euler = row.get("euler_error", {})
    result = {
        "case_id": row["case_id"],
        "mode": mode_id(row["mode"]),
        "step": int(row["step"]),
        "full_step": full,
        "active_count": int(row["active_count"]),
        "inactive_count": int(row["inactive_count"]),
        "active_fraction": row["active_count"] / (row["active_count"] + row["inactive_count"]),
        "sigma_delta": delta,
        "current_latent_sha256": row.get("current_latent_sha256"),
        "active_error_energy_fraction_computed": a / (a + i) if a + i else None,
        "inactive_error_energy_fraction_computed": i / (a + i) if a + i else None,
        "inactive_error_energy_fraction_recorded": row.get("inactive_error_energy_fraction"),
        "sse_additivity_residual": residual,
        "sse_additivity_pass": abs(residual) <= tolerance,
        "orthogonality_cross_term": cross,
        "orthogonality_pass": abs(float(cross)) <= tolerance if finite(cross) else None,
        "factorization_max_abs": row.get("factorization_max_abs"),
        "factorization_exact_pass": row.get("factorization_max_abs") == 0 if row.get("factorization_max_abs") is not None else None,
        "actual_euler_update_bitwise_equal": row.get("actual_euler_update_bitwise_equal"),
        "oracle_sampled_cache_unchanged": row.get("oracle_sampled_cache_unchanged"),
        "euler_active_squared_error_sum": delta * delta * a,
        "euler_inactive_squared_error_sum": delta * delta * i,
        "euler_combined_squared_error_sum": delta * delta * total,
        "euler_active_rmse_recorded": euler.get("active_rmse"),
        "euler_inactive_rmse_recorded": euler.get("inactive_rmse"),
        "euler_combined_rmse_recorded": euler.get("combined_rmse"),
        "full_step_exact_zero_error": a == 0 and i == 0 and total == 0 if full else None,
        "returned_sse_fraction_of_raw_same_state_sse": float(returned["squared_error_sum"]) / total if returned.get("squared_error_sum") is not None and total else None,
        "output_mixing_sse_identity_pass": abs(float(returned["squared_error_sum"]) - expected_returned_sse) <= tolerance if returned.get("squared_error_sum") is not None and expected_returned_sse is not None else None,
        "local_seconds_research_only": row.get("local_seconds"),
        "oracle_seconds_research_only": row.get("oracle_seconds"),
    }
    for key in ERROR_FIELDS:
        value = returned.get(key)
        result[f"returned_{key}"] = value if value is None or finite(value) else None
        rounded_value = row.get("rounded_euler_error", {}).get(key)
        result[f"rounded_euler_{key}"] = rounded_value if rounded_value is None or finite(rounded_value) else None
    rounded_rmse = row.get("rounded_euler_error", {}).get("rmse")
    result["rounded_euler_rmse_minus_analytic"] = float(rounded_rmse) - float(euler["combined_rmse"]) if finite(rounded_rmse) and finite(euler.get("combined_rmse")) else None
    for region, values in [("active", active), ("inactive", inactive), ("combined", combined)]:
        for key in ERROR_FIELDS:
            value = values.get(key)
            result[f"{region}_{key}"] = value if value is None or finite(value) else None
        recorded = euler.get(f"{region}_rmse")
        expected = abs(delta) * float(values["rmse"]) if finite(values.get("rmse")) else None
        result[f"euler_{region}_rmse_from_velocity"] = expected
        result[f"euler_{region}_analytic_rmse_consistent"] = bool(math.isclose(float(recorded), expected, rel_tol=1e-4, abs_tol=1e-7)) if finite(recorded) and expected is not None else None
    return result


def aggregate_steps(rows, case_id, mode):
    selected = [row for row in rows if row["case_id"] == case_id and row["mode"] == mode]
    partial = [row for row in selected if not row["full_step"]]
    a = sum(row["active_squared_error_sum"] for row in partial)
    i = sum(row["inactive_squared_error_sum"] for row in partial)
    ea = sum(row["euler_active_squared_error_sum"] for row in partial)
    ei = sum(row["euler_inactive_squared_error_sum"] for row in partial)
    return {
        "case_id": case_id,
        "mode": mode,
        "recorded_steps": [row["step"] for row in selected],
        "partial_steps": [row["step"] for row in partial],
        "partial_step_count": len(partial),
        "active_sse_sum_over_recorded_partial_steps": a,
        "inactive_sse_sum_over_recorded_partial_steps": i,
        "active_share_of_summed_partial_velocity_sse": a / (a + i) if a + i else None,
        "inactive_share_of_summed_partial_velocity_sse": i / (a + i) if a + i else None,
        "active_share_of_summed_one_step_euler_sse": ea / (ea + ei) if ea + ei else None,
        "inactive_share_of_summed_one_step_euler_sse": ei / (ea + ei) if ea + ei else None,
        "summed_one_step_euler_sse": ea + ei,
        "full_step_count": sum(row["full_step"] for row in selected),
        "full_step_exact_zero_all": all(row["full_step_exact_zero_error"] for row in selected if row["full_step"]) if any(row["full_step"] for row in selected) else None,
        "sse_additivity_all": all(row["sse_additivity_pass"] for row in selected) if selected else None,
        "orthogonality_all": all(row["orthogonality_pass"] for row in selected if row["orthogonality_pass"] is not None) if selected else None,
    }


def renamed_quality(values):
    result = {}
    for key, value in values.items():
        name = key.replace("_to_F", "_to_D").replace("F_far_background", "D_far_background").replace("_over_F", "_over_D")
        result[name] = value
    return result
